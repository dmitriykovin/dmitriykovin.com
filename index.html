  <script>

    function openLightbox(el) {

      document.getElementById('lightbox-img').src = el.src;

      document.getElementById('lightbox').classList.remove('hidden');

      document.getElementById('lightbox').classList.add('flex');

    }


    function openVideo(video) {

      const lb = document.getElementById('lightbox');

      lb.innerHTML = `

        <div class="relative max-w-[95%] max-h-[95vh]">

          <button onclick="closeLightbox()" class="absolute -top-6 right-4 text-5xl text-white hover:text-purple-400">×</button>

          ${video.outerHTML}

        </div>`;

      lb.classList.remove('hidden');

      lb.classList.add('flex');

      lb.querySelector('video').play();

    }


    function closeLightbox() {

      const lb = document.getElementById('lightbox');

      lb.classList.add('hidden');

      lb.classList.remove('flex');

    }

  </script>
